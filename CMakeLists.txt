cmake_minimum_required(VERSION 3.16)
project(ShadingHelp LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
add_library(shading-core src/core.cpp src/progressive.cpp)
target_link_libraries(shading-core PUBLIC Threads::Threads)
if(MSVC)
  target_compile_options(shading-core PRIVATE /W4 /utf-8)
else()
  target_compile_options(shading-core PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(shading-render src/cli.cpp)
target_link_libraries(shading-render PRIVATE shading-core)
add_executable(shading-tests tests/tests.cpp)
target_link_libraries(shading-tests PRIVATE shading-core)
enable_testing()
add_test(NAME numerical-and-renderer COMMAND shading-tests)
if(WIN32)
  add_executable(ShadingHelp WIN32 src/windows.cpp)
  target_compile_definitions(ShadingHelp PRIVATE _WIN32_WINNT=0x0A00)
  target_link_libraries(ShadingHelp PRIVATE shading-core user32 gdi32 comctl32 comdlg32 shell32)
  if(MINGW)
    target_link_options(ShadingHelp PRIVATE -municode -static)
  endif()
endif()
